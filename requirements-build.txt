# BUILD-TIME dependencies — NOT installed in the deployed image.

# Needed to export SPECTER2 to ONNX (pipeline/13_export_onnx.py), build the

-r requirements.txt

# CPU torch is enough for exporting and index building. Install it from the CPU
# index or pip pulls ~2.5GB of CUDA that will never run:
#   pip install torch==2.13.0 --index-url https://download.pytorch.org/whl/cpu
torch==2.13.0
# 4.57.6, NOT 5.x: optimum-onnx (pulled by optimum[onnxruntime]) requires
# transformers<4.58.0, and a fresh resolve with 5.x is ResolutionImpossible —
# pip exits in seconds, which is exactly how the a5ee046 deploy died. The pin
# originally said 5.14.1 because that was in the venv when the file was
# written; installing optimum afterwards silently downgraded it to 4.57.6,
# so 4.57.6 is what every local validation actually ran against.
transformers==4.57.6
sentence-transformers==5.6.0

# ONNX export + dynamic quantization
optimum[onnxruntime]==2.1.0

# Test runner
pytest
